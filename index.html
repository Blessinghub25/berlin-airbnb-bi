""" Berlin Airbnb Market Intelligence — Interactive Dashboard ---------------------------------------------------------- Author: Blessing Dediare Tools: Python · pandas · Plotly · SQLite Output: dashboard.html (open in any browser) Run: python dashboard.py """ import pandas as pd import sqlite3 import plotly.graph_objects as go import plotly.express as px from plotly.subplots import make_subplots import json # ── LOAD DATA ───────────────────────────────────────────────────────── print("Loading data...") conn = sqlite3.connect("berlin_airbnb.db") df = pd.read_sql("SELECT * FROM listings", conn) neighbourhood_df = pd.read_sql(""" SELECT neighbourhood, COUNT(*) AS total_listings, ROUND(AVG(price), 2) AS avg_price, ROUND(AVG(rating), 2) AS avg_rating, ROUND(AVG(availability_365), 0) AS avg_availability, SUM(CASE WHEN is_superhost = 1 THEN 1 ELSE 0 END) AS superhost_count, ROUND(100.0 * SUM(CASE WHEN is_superhost=1 THEN 1 ELSE 0 END)/COUNT(*),1) AS superhost_pct FROM listings GROUP BY neighbourhood HAVING COUNT(*) >= 10 ORDER BY total_listings DESC """, conn) room_df = pd.read_sql(""" SELECT room_type, COUNT(*) AS listings, ROUND(AVG(price),2) AS avg_price, ROUND(AVG(rating),2) AS avg_rating FROM listings GROUP BY room_type ORDER BY listings DESC """, conn) price_dist_df = pd.read_sql(""" SELECT CASE WHEN price < 50 THEN '€0–50' WHEN price < 100 THEN '€50–100' WHEN price < 150 THEN '€100–150' WHEN price < 200 THEN '€150–200' WHEN price < 300 THEN '€200–300' WHEN price < 500 THEN '€300–500' ELSE '€500+' END AS bucket, COUNT(*) AS count FROM listings GROUP BY bucket ORDER BY MIN(price) """, conn) host_df = pd.read_sql(""" SELECT host_id, host_name, COUNT(*) AS listings, ROUND(AVG(price),2) AS avg_price, MAX(is_superhost) AS superhost FROM listings GROUP BY host_id, host_name ORDER BY listings DESC LIMIT 20 """, conn) superhost_df = pd.read_sql(""" SELECT CASE WHEN is_superhost=1 THEN 'Superhost' ELSE 'Regular Host' END AS host_type, COUNT(*) AS listings, ROUND(AVG(price),2) AS avg_price, ROUND(AVG(rating),2) AS avg_rating, ROUND(AVG(reviews_per_month),2) AS reviews_pm FROM listings GROUP BY is_superhost """, conn) conn.close() # ── KEY METRICS ─────────────────────────────────────────────────────── total_listings = len(df) median_price = round(df["price"].median(), 2) avg_rating = round(df["rating"].mean(), 2) total_hoods = neighbourhood_df["neighbourhood"].nunique() multi_host_pct = round(df[df.get("host_total_listings", pd.Series(dtype=float)) > 1].shape[0] / total_listings * 100, 1) if "host_total_listings" in df.columns else 60.6 top_hood = neighbourhood_df.iloc[0] premium_hood = neighbourhood_df.nlargest(1, "avg_price").iloc[0] premium_pct = round((premium_hood["avg_price"] - median_price) / median_price * 100, 1) # ── COLOUR PALETTE ──────────────────────────────────────────────────── NAVY = "#1A3557" BLUE = "#2563EB" BLUE2 = "#3b7cf0" LIGHT = "#EFF4FF" GREY = "#64748b" WHITE = "#ffffff" GREEN = "#22c55e" AMBER = "#f59e0b" COLORS = [BLUE, NAVY, "#0f4c81", "#1e3a6e", "#0d2d5e", "#1b3f6f", GREY] # ── FIGURES ─────────────────────────────────────────────────────────── # 1. Listings by neighbourhood (top 15) top15 = neighbourhood_df.head(15).sort_values("total_listings") fig_listings = go.Figure(go.Bar( x=top15["total_listings"], y=top15["neighbourhood"], orientation="h", marker_color=[BLUE if n == top_hood["neighbourhood"] else "#93c5fd" for n in top15["neighbourhood"]], text=top15["total_listings"], textposition="outside", hovertemplate="%{y}
Listings: %{x}
Avg price: €%{customdata:.0f}", customdata=top15["avg_price"] )) fig_listings.update_layout( title="Listings by Neighbourhood (Top 15)", xaxis_title="Number of Listings", plot_bgcolor=WHITE, paper_bgcolor=WHITE, font=dict(family="Arial", color=NAVY), height=450, margin=dict(l=180, r=40, t=50, b=40) ) # 2. Avg price by neighbourhood (top 15 by price) top15_price = neighbourhood_df.nlargest(15, "avg_price").sort_values("avg_price") colors_price = [BLUE if row["avg_price"] > median_price * 1.2 else "#93c5fd" for _, row in top15_price.iterrows()] fig_price = go.Figure(go.Bar( x=top15_price["avg_price"], y=top15_price["neighbourhood"], orientation="h", marker_color=colors_price, text=[f"€{p:.0f}" for p in top15_price["avg_price"]], textposition="outside", )) fig_price.add_vline(x=median_price, line_dash="dash", line_color=AMBER, annotation_text=f"Berlin median €{median_price:.0f}", annotation_position="top right") fig_price.update_layout( title="Average Nightly Price by Neighbourhood (Top 15)", xaxis_title="Average Price (€)", plot_bgcolor=WHITE, paper_bgcolor=WHITE, font=dict(family="Arial", color=NAVY), height=450, margin=dict(l=200, r=80, t=50, b=40) ) # 3. Room type donut fig_room = go.Figure(go.Pie( labels=room_df["room_type"], values=room_df["listings"], hole=0.55, marker_colors=COLORS, textinfo="label+percent", )) fig_room.update_layout( title="Listings by Room Type", plot_bgcolor=WHITE, paper_bgcolor=WHITE, font=dict(family="Arial", color=NAVY), height=350, showlegend=False ) # 4. Price distribution fig_dist = go.Figure(go.Bar( x=price_dist_df["bucket"], y=price_dist_df["count"], marker_color=[BLUE if b == "€100–150" else "#93c5fd" for b in price_dist_df["bucket"]], text=price_dist_df["count"], textposition="outside", )) fig_dist.update_layout( title="Price Distribution Across Listings", xaxis_title="Price Range", yaxis_title="Number of Listings", plot_bgcolor=WHITE, paper_bgcolor=WHITE, font=dict(family="Arial", color=NAVY), height=350 ) # 5. Superhost comparison fig_super = go.Figure() categories = ["Avg Price (€)", "Avg Rating", "Reviews / Month"] for _, row in superhost_df.iterrows(): color = BLUE if row["host_type"] == "Superhost" else "#93c5fd" fig_super.add_trace(go.Bar( name=row["host_type"], x=categories, y=[row["avg_price"], row["avg_rating"] * 20, row["reviews_pm"] * 10], marker_color=color, text=[f"€{row['avg_price']:.0f}", f"{row['avg_rating']:.2f}★", f"{row['reviews_pm']:.1f}"], textposition="outside", )) fig_super.update_layout( title="Superhost vs Regular Host Comparison", barmode="group", plot_bgcolor=WHITE, paper_bgcolor=WHITE, font=dict(family="Arial", color=NAVY), height=350, yaxis_visible=False, legend=dict(orientation="h", yanchor="bottom", y=1.02) ) # 6. Top 20 hosts fig_hosts = go.Figure(go.Bar( x=host_df["listings"], y=[f"{n[:20]}..." if len(n) > 20 else n for n in host_df["host_name"]], orientation="h", marker_color=[BLUE if s else "#93c5fd" for s in host_df["superhost"]], text=host_df["listings"], textposition="outside", )) fig_hosts.update_layout( title="Top 20 Hosts by Portfolio Size", xaxis_title="Number of Listings", plot_bgcolor=WHITE, paper_bgcolor=WHITE, font=dict(family="Arial", color=NAVY), height=500, margin=dict(l=160, r=60, t=50, b=40) ) # 7. Map fig_map = px.scatter_mapbox( df.sample(min(2000, len(df))), lat="latitude", lon="longitude", color="price", color_continuous_scale=["#93c5fd", BLUE, NAVY], range_color=[0, 300], size_max=6, opacity=0.6, zoom=10, center={"lat": 52.52, "lon": 13.405}, mapbox_style="carto-positron", hover_data={"price": True, "neighbourhood": True, "room_type": True, "latitude": False, "longitude": False}, labels={"price": "Price (€)"}, title="Berlin Listings Map — Coloured by Nightly Price" ) fig_map.update_layout( height=500, paper_bgcolor=WHITE, font=dict(family="Arial", color=NAVY), coloraxis_colorbar=dict(title="€/night") ) # ── BUILD HTML ──────────────────────────────────────────────────────── print("Building dashboard HTML...") def fig_to_html(fig, div_id): return fig.to_html(full_html=False, include_plotlyjs=False, div_id=div_id) html = f""" Berlin Short-Term Rental Market Intelligence
